from pyscript import display from pyodide.http import pyfetch display("Hello, World!") print("Debugging: script started") url = "https://jsonplaceholder.typicode.com/posts" response = await pyfetch(url) posts = await response.json() for post in posts[:5]: display(f"Post ID: {post['id']}, Title: {post['title']}") print(f"Debugging - Post ID: {post['id']}, Title: {post['title']}") print("Debugging: script finished")
